# h2open desktop app: C++23 and Qt 6 Widgets.
#
#   cmake --preset dev && cmake --build --preset dev && ctest --preset dev
#
# Presets (CMakePresets.json): dev (Clang, ASan + UBSan), tsan (Clang,
# ThreadSanitizer), fuzz (Clang, libFuzzer), release (default compiler).
cmake_minimum_required(VERSION 3.28)
project(h2open-desktop VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# No C++ modules: skip the module scan (and its GCC-only flags).
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
# clang-tidy and clazy read the compile commands.
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(H2OPEN_SANITIZE "" CACHE STRING
    "Sanitizers for every target, e.g. address,undefined or thread")
option(H2OPEN_FUZZ "Build the libFuzzer harnesses (Clang only)" OFF)

find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Network Test)
qt_standard_project_setup(REQUIRES 6.8)

# --- Options for our code (Qt's headers are system headers, so their
# --- warnings stay out). -----------------------------------------------------
add_library(h2open_options INTERFACE)
if(MSVC)
    target_compile_options(h2open_options INTERFACE /W4 /WX /permissive-)
else()
    target_compile_options(h2open_options INTERFACE
        -Wall -Wextra -Wpedantic -Werror
        -Wconversion -Wsign-conversion -Wshadow -Wold-style-cast
        -Wnon-virtual-dtor -Woverloaded-virtual -Wnull-dereference
        -Wimplicit-fallthrough -Wcast-qual -Wformat=2)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    # Safe Buffers: no raw pointer arithmetic or indexing; use std::span,
    # containers and QStringView. Only Clang has it.
    target_compile_options(h2open_options INTERFACE -Wunsafe-buffer-usage)
endif()
target_compile_definitions(h2open_options INTERFACE
    # Hardened standard library: out-of-range [], front() on an empty
    # container, *empty_optional... abort instead of being silent UB.
    # libstdc++ (GCC, and Clang on Linux):
    _GLIBCXX_ASSERTIONS
    # libc++ (Clang on macOS, or with -stdlib=libc++):
    _LIBCPP_HARDENING_MODE=_LIBCPP_HARDENING_MODE_EXTENSIVE
    # Every string must say its encoding (u"…"_s), no narrowing in connect(),
    # nothing deprecated up to Qt 6.8. (QT_STRICT_ITERATORS was Qt 5 only.)
    QT_NO_CAST_FROM_ASCII
    QT_NO_CAST_TO_ASCII
    QT_NO_URL_CAST_FROM_STRING
    QT_NO_NARROWING_CONVERSIONS_IN_CONNECT
    QT_DISABLE_DEPRECATED_UP_TO=0x060800)

if(H2OPEN_SANITIZE)
    # Applied to everything we build, so libraries and tests agree. Any
    # finding stops the program (no "recover and continue").
    add_compile_options(-fsanitize=${H2OPEN_SANITIZE} -fno-omit-frame-pointer
                        -fno-sanitize-recover=all)
    add_link_options(-fsanitize=${H2OPEN_SANITIZE})
    if(H2OPEN_SANITIZE MATCHES "address")
        add_compile_options(-fsanitize-address-use-after-scope)
    endif()
endif()
if(H2OPEN_FUZZ)
    # Instrument our libraries for coverage-guided fuzzing.
    add_compile_options(-fsanitize=fuzzer-no-link)
endif()

# Code generated by moc is not ours and indexes raw pointer arrays by design:
# Safe Buffers is turned off for those files only.
function(h2open_target target)
    target_link_libraries(${target} PRIVATE h2open_options)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        set_source_files_properties(
            ${CMAKE_CURRENT_BINARY_DIR}/${target}_autogen/mocs_compilation.cpp
            TARGET_DIRECTORY ${target}
            PROPERTIES COMPILE_OPTIONS "-Wno-unsafe-buffer-usage")
    endif()
endfunction()

# --- Libraries -------------------------------------------------------------
# Talking to the server: no widgets, so it is fast to test and fuzz.
add_library(h2open_api STATIC src/api.cpp src/api.h)
target_include_directories(h2open_api PUBLIC src)
target_link_libraries(h2open_api PUBLIC Qt6::Core Qt6::Network)
h2open_target(h2open_api)

# The windows, apart from main() so tests can create them.
add_library(h2open_ui STATIC
    src/appearance.cpp src/appearance.h
    src/desktopapp.cpp src/desktopapp.h
    src/dimlabel.cpp src/dimlabel.h
    src/gallery.cpp src/gallery.h
    src/loginwindow.cpp src/loginwindow.h
    src/mainwindow.cpp src/mainwindow.h
    src/qtmemory.h)
target_link_libraries(h2open_ui PUBLIC h2open_api Qt6::Widgets)
h2open_target(h2open_ui)

qt_add_executable(h2open-desktop WIN32 src/main.cpp)
target_link_libraries(h2open-desktop PRIVATE h2open_ui)
h2open_target(h2open-desktop)

# --- Tests: Qt Test, without a display -------------------------------------
enable_testing()
foreach(name IN ITEMS api windows)
    qt_add_executable(test_${name} tests/test_${name}.cpp)
    target_link_libraries(test_${name} PRIVATE h2open_ui Qt6::Test)
    h2open_target(test_${name})
    add_test(NAME ${name} COMMAND test_${name})
    set_tests_properties(${name} PROPERTIES ENVIRONMENT
        "QT_QPA_PLATFORM=offscreen;DBUS_SESSION_BUS_ADDRESS=disabled:;ASAN_OPTIONS=detect_leaks=1:strict_string_checks=1;UBSAN_OPTIONS=print_stacktrace=1;TSAN_OPTIONS=suppressions=${CMAKE_SOURCE_DIR}/tests/tsan.supp:halt_on_error=1")
endforeach()

# --- Fuzzing: anything that reads network input ------------------------------
if(H2OPEN_FUZZ)
    add_executable(fuzz_parse_session fuzz/parse_session.cpp)
    target_link_libraries(fuzz_parse_session PRIVATE h2open_api)
    h2open_target(fuzz_parse_session)
    target_link_options(fuzz_parse_session PRIVATE -fsanitize=fuzzer)
endif()

# --- clazy: Qt-specific lints (cmake --build --preset dev --target clazy) ---
set(H2OPEN_CLAZY_CHECKS
    "level0,level1,level2,container-inside-loop,heap-allocated-small-trivial-type,isempty-vs-count,signal-with-return-value,thread-with-slots,unneeded-cast,use-arrow-operator-instead-of-data")
file(GLOB H2OPEN_SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/src/*.cpp
     ${CMAKE_SOURCE_DIR}/tests/*.cpp)
add_custom_target(clazy
    COMMAND clazy-standalone -p ${CMAKE_BINARY_DIR} --checks=${H2OPEN_CLAZY_CHECKS}
            "--header-filter=.*/desktop/(src|tests)/.*" --extra-arg=-Werror
            ${H2OPEN_SOURCES}
    WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
    COMMENT "clazy: Qt lints"
    VERBATIM)
